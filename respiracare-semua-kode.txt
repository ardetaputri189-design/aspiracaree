==================== FILE: README.md ====================
# RespiraCare

React + TypeScript + Tailwind + Recharts + Leaflet. Run:

```bash
npm install
cp .env.example .env
npm run dev
```

- Air quality, weather, and place search use Open-Meteo (no API key).
- Surroundings use OpenStreetMap (Overpass API).
- Map station markers need `VITE_OPENAQ_KEY`.
- Scan logging to Supabase is optional: run `supabase/schema.sql`, then set the two Supabase variables.
- `VITE_DEMO_MODE=true` shows clearly labelled sample data.
- The `components/ui` files follow shadcn/ui conventions (cva + tailwind-merge). To use the full library, run `npx shadcn@latest init`.

import { AQI_NAME, getLevel } from '@/lib/aqi'
export function AQIGauge({ value, size = 220 }: { value: number | null; size?: number }) {
  const r = 80, full = 2 * Math.PI * r, arc = full * 0.75
  const level = value == null ? null : getLevel(value)
  const frac = Math.min(value ?? 0, 500) / 500
  return (
    <figure className="m-0 text-center" aria-label={value == null ? `${AQI_NAME} not available` : `${AQI_NAME} ${value}, ${level!.label}`}>
      <svg width={size} height={size} viewBox="0 0 200 200" role="img">
        <circle cx="100" cy="100" r={r} fill="none" stroke="#E2E8F0" strokeWidth="14" strokeLinecap="round" strokeDasharray={`${arc} ${full}`} transform="rotate(135 100 100)" />
        {level && <circle cx="100" cy="100" r={r} fill="none" stroke={level.color} strokeWidth="14" strokeLinecap="round" strokeDasharray={`${Math.max(arc * frac, 1)} ${full}`} transform="rotate(135 100 100)" style={{ transition: 'stroke-dasharray .6s ease' }} />}
        <text x="100" y="102" textAnchor="middle" fontSize="44" fontWeight="700" fill="#172B3A">{value == null ? '–' : Math.round(value)}</text>
        <text x="100" y="126" textAnchor="middle" fontSize="12" fill="#64748B">{AQI_NAME}</text>
      </svg>
      {level && <figcaption className="-mt-4 inline-block rounded-full px-3 py-1 text-sm font-semibold text-white" style={{ background: level.color }}>{level.label}</figcaption>}
    </figure>
  )
}

import { useEffect, useState } from 'react'
import { CircleMarker, MapContainer, Popup, TileLayer, useMap, useMapEvents } from 'react-leaflet'
import 'leaflet/dist/leaflet.css'
import { fetchStationLatest, fetchStations, hasStations, type Place, type Station, type StationReading } from '@/lib/api'
import { pm25Band } from '@/lib/aqi'

function Recenter({ p }: { p: Place }) { const map = useMap(); useEffect(() => { map.setView([p.lat, p.lon], Math.max(map.getZoom(), 11)) }, [p.lat, p.lon]); return null }
function Picker({ onPick }: { onPick: (lat: number, lon: number) => void }) { useMapEvents({ click: e => onPick(e.latlng.lat, e.latlng.lng) }); return null }

function StationMarker({ s }: { s: Station }) {
  const [r, setR] = useState<StationReading[] | null>(null)
  const [err, setErr] = useState(false)
  const pm = r?.find(x => x.parameter === 'pm25')
  const band = pm ? pm25Band(pm.value) : null
  const load = () => { if (r) return; fetchStationLatest(s).then(setR).catch(() => setErr(true)) }
  return (
    <CircleMarker center={[s.lat, s.lon]} radius={9} pathOptions={{ color: '#fff', weight: 2, fillColor: band?.color ?? '#94A3B8', fillOpacity: 1 }} eventHandlers={{ popupopen: load }}>
      <Popup>
        <strong>{s.name}</strong><br /><span style={{ color: '#64748B' }}>{s.provider}</span>
        <div style={{ marginTop: 6 }}>
          {err ? 'Latest readings are unavailable.' : !r ? 'Loading latest readings…' : r.length === 0 ? 'No recent readings.' : (
            <>{pm && <div><b>PM2.5: {pm.value} {pm.units}</b>{band && <> (band: {band.label})</>}</div>}
              {r.filter(x => x.parameter !== 'pm25').slice(0, 4).map(x => <div key={x.parameter}>{x.parameter.toUpperCase()}: {x.value} {x.units}</div>)}
              <div style={{ color: '#64748B', fontSize: 11 }}>{r[0].time}</div></>
          )}
        </div>
        <div style={{ marginTop: 6, fontSize: 11, color: '#64748B' }}>This reading represents the station location only.</div>
      </Popup>
    </CircleMarker>
  )
}

export function InteractiveMap({ place, onPick }: { place: Place | null; onPick: (lat: number, lon: number) => void }) {
  const [stations, setStations] = useState<Station[] | null>(null)
  const [stErr, setStErr] = useState(false)
  useEffect(() => {
    setStations(null); setStErr(false)
    if (place && hasStations) fetchStations(place.lat, place.lon).then(setStations).catch(() => setStErr(true))
  }, [place?.lat, place?.lon])
  const center: [number, number] = place ? [place.lat, place.lon] : [-2.5, 118]
  return (
    <div>
      <div className="h-[380px] overflow-hidden rounded-2xl border border-slate-100 shadow-card">
        <MapContainer center={center} zoom={place ? 11 : 4} scrollWheelZoom={false} className="h-full w-full">
          <TileLayer attribution='&copy; <a href="https://www.openstreetmap.org/copyright">OpenStreetMap</a> contributors' url="https://tile.openstreetmap.org/{z}/{x}/{y}.png" />
          <Picker onPick={onPick} />
          {place && <><Recenter p={place} />
            <CircleMarker center={[place.lat, place.lon]} radius={10} pathOptions={{ color: '#fff', weight: 3, fillColor: '#087F8C', fillOpacity: 1 }}><Popup><strong>{place.name}</strong><br />Selected location</Popup></CircleMarker></>}
          {stations?.map(s => <StationMarker key={s.id} s={s} />)}
        </MapContainer>
      </div>
      <p className="mt-2 text-xs text-ink-muted" aria-live="polite">
        {!hasStations ? 'Click the map to choose a location. Monitoring-station markers appear when an OpenAQ API key (VITE_OPENAQ_KEY) is configured.'
          : stErr ? 'Monitoring stations could not be loaded right now.'
          : !place ? 'Choose a location to see monitoring stations within 25 km.'
          : stations === null ? 'Loading monitoring stations…'
          : stations.length === 0 ? 'No monitoring stations found within 25 km. Values on this page are model estimates.'
          : `${stations.length} monitoring stations within 25 km. Grey markers have not loaded readings yet; click one to load. Colours show PM2.5 bands, and each reading represents only its own station.`}
      </p>
    </div>
  )
}

import { useEffect, useState } from 'react'
import { Crosshair, Loader2, MapPin, Search } from 'lucide-react'
import { geocode, placeLabel, type Place } from '@/lib/api'
import { useLocationData } from '@/context/LocationContext'
import { Button } from '@/components/ui/button'

export function LocationSearch() {
  const { select, locate, locating, locateError } = useLocationData()
  const [q, setQ] = useState('')
  const [results, setResults] = useState<Place[]>([])
  const [busy, setBusy] = useState(false)
  const [msg, setMsg] = useState<string | null>(null)
  useEffect(() => {
    if (q.trim().length < 2) { setResults([]); setMsg(null); return }
    const id = setTimeout(async () => {
      setBusy(true)
      try { const r = await geocode(q.trim()); setResults(r); setMsg(r.length ? null : 'No matching places. Try a nearby city name.') }
      catch { setMsg('Search is unavailable right now. Check your connection and try again.') }
      finally { setBusy(false) }
    }, 350)
    return () => clearTimeout(id)
  }, [q])
  const pick = (p: Place) => { setQ(''); setResults([]); select(p) }
  return (
    <div>
      <div className="flex flex-col gap-3 sm:flex-row">
        <div className="relative flex-1">
          <Search size={18} className="absolute left-3 top-3.5 text-ink-muted" />
          <input value={q} onChange={e => setQ(e.target.value)} onKeyDown={e => { if (e.key === 'Enter' && results[0]) pick(results[0]) }} placeholder="Search a city or place, e.g. Malang" aria-label="Search location" className="h-11 w-full rounded-xl border border-slate-200 bg-white pl-10 pr-10 text-sm" />
          {busy && <Loader2 size={18} className="absolute right-3 top-3.5 animate-spin text-ink-muted" />}
          {results.length > 0 && (
            <ul className="absolute z-[1000] mt-1 w-full overflow-hidden rounded-xl border border-slate-100 bg-white shadow-card">
              {results.map(p => <li key={`${p.lat},${p.lon}`}><button className="flex w-full items-center gap-2 px-3 py-2.5 text-left text-sm hover:bg-slate-50" onClick={() => pick(p)}><MapPin size={16} className="text-brand" />{placeLabel(p)}</button></li>)}
            </ul>
          )}
        </div>
        <Button variant="outline" onClick={locate} disabled={locating}>{locating ? <Loader2 size={18} className="animate-spin" /> : <Crosshair size={18} />}Use My Location</Button>
      </div>
      <p className="mt-2 min-h-5 text-sm text-ink-muted" aria-live="polite">{msg ?? locateError}</p>
    </div>
  )
}

const lerp = (a: number, b: number, t: number) => a + (b - a) * t
function mix(c1: string, c2: string, t: number) {
  const p = (c: string) => [1, 3, 5].map(i => parseInt(c.slice(i, i + 2), 16))
  const a = p(c1), b = p(c2)
  return '#' + a.map((v, i) => Math.round(lerp(v, b[i], t)).toString(16).padStart(2, '0')).join('')
}

import { createContext, useCallback, useContext, useEffect, useRef, useState, type ReactNode } from 'react'
import { fetchAir, reverseName, placeLabel, type AirData, type Place } from '@/lib/api'
import { detectEnvironment, type EnvResult } from '@/lib/environment'
import { saveScan } from '@/lib/supabase'

type Status = 'idle' | 'loading' | 'ready' | 'error'
interface Ctx {
  place: Place | null; air: AirData | null; env: EnvResult | null; envError: string | null
  status: Status; error: string | null; locating: boolean; locateError: string | null
  select: (p: Place) => void; locate: () => void; sensitive: boolean; setSensitive: (v: boolean) => void
}
const LocationCtx = createContext<Ctx | null>(null)
const KEY = 'respiracare:place'

export function LocationProvider({ children }: { children: ReactNode }) {
  const [place, setPlace] = useState<Place | null>(null)
  const [air, setAir] = useState<AirData | null>(null)
  const [env, setEnv] = useState<EnvResult | null>(null)
  const [envError, setEnvError] = useState<string | null>(null)
  const [status, setStatus] = useState<Status>('idle')
  const [error, setError] = useState<string | null>(null)
  const [locating, setLocating] = useState(false)
  const [locateError, setLocateError] = useState<string | null>(null)
  const [sensitive, setSensitive] = useState(false)
  const req = useRef(0)

  const select = useCallback((p: Place) => {
    const id = ++req.current
    setPlace(p); setStatus('loading'); setError(null); setEnvError(null); setAir(null); setEnv(null)
    try { localStorage.setItem(KEY, JSON.stringify(p)) } catch { /* ignore */ }
    Promise.allSettled([fetchAir(p), detectEnvironment(p)]).then(([a, e]) => {
      if (id !== req.current) return
      if (a.status === 'rejected') { setStatus('error'); setError('Could not load air quality data for this location. Check your connection and try again.') }
      else { setAir(a.value); setStatus('ready') }
      if (e.status === 'fulfilled') setEnv(e.value); else setEnvError('Environment data from OpenStreetMap is unavailable right now.')
      if (a.status === 'fulfilled') saveScan({ place: placeLabel(p), lat: p.lat, lon: p.lon, aqi: a.value.aqi, pm25: a.value.pm25, category: e.status === 'fulfilled' ? e.value.category : null, demo: a.value.isDemo })
    })
  }, [])

  const locate = useCallback(() => {
    setLocateError(null)
    if (!navigator.geolocation) { setLocateError('Your browser does not support location access. Search for a city instead.'); return }
    setLocating(true)
    navigator.geolocation.getCurrentPosition(
      async pos => { const p = await reverseName(pos.coords.latitude, pos.coords.longitude); setLocating(false); select(p) },
      () => { setLocating(false); setLocateError('Location permission was denied or unavailable. Search for a city instead.') },
      { timeout: 10000 }
    )
  }, [select])

  useEffect(() => {
    try { const s = localStorage.getItem(KEY); if (s) select(JSON.parse(s) as Place) } catch { /* ignore */ }
  }, [select])

  return <LocationCtx.Provider value={{ place, air, env, envError, status, error, locating, locateError, select, locate, sensitive, setSensitive }}>{children}</LocationCtx.Provider>
}
export function useLocationData() {
  const c = useContext(LocationCtx)
  if (!c) throw new Error('useLocationData must be used inside LocationProvider')
  return c
}

export interface Place { name: string; admin?: string; country?: string; lat: number; lon: number }

export interface AirData {
  isDemo: boolean; source: string; timezone: string; time: string
  aqi: number | null; pm25: number | null; pm10: number | null; co: number | null; no2: number | null; o3: number | null
  temp: number | null; humidity: number | null; wind: number | null; history: HistoryPoint[]
}
export const placeLabel = (p: Place) => [p.name, p.admin, p.country].filter(Boolean).join(', ')
const num = (v: unknown): number | null => (typeof v === 'number' && isFinite(v) ? v : null)
async function getJson(url: string, init?: RequestInit) {
  const r = await fetch(url, init)
  if (!r.ok) throw new Error(`Request failed (${r.status})`)
  return r.json()
}

export async function geocode(q: string): Promise<Place[]> {
  const j = await getJson(`https://geocoding-api.open-meteo.com/v1/search?name=${encodeURIComponent(q)}&count=6&language=en&format=json`)
  return (j.results ?? []).map((x: any) => ({ name: x.name, admin: x.admin1, country: x.country, lat: x.latitude, lon: x.longitude }))
}
export async function reverseName(lat: number, lon: number): Promise<Place> {
  try {
    const j = await getJson(`https://nominatim.openstreetmap.org/reverse?format=jsonv2&zoom=10&lat=${lat}&lon=${lon}`)
    const a = j.address ?? {}
    return { name: a.city || a.town || a.village || a.county || j.name || 'Selected location', admin: a.state, country: a.country, lat, lon }
  } catch { return { name: `${lat.toFixed(3)}, ${lon.toFixed(3)}`, lat, lon } }
}

export async function fetchAir(p: Place): Promise<AirData> {
  if (import.meta.env.VITE_DEMO_MODE === 'true') return demoAir()
  const q = `latitude=${p.lat}&longitude=${p.lon}&timezone=auto`
  const [air, wx] = await Promise.all([
    getJson(`https://air-quality-api.open-meteo.com/v1/air-quality?${q}&current=us_aqi,pm10,pm2_5,carbon_monoxide,nitrogen_dioxide,ozone&hourly=us_aqi,pm10,pm2_5&past_days=1&forecast_days=1`),
    getJson(`https://api.open-meteo.com/v1/forecast?${q}&current=temperature_2m,relative_humidity_2m,wind_speed_10m`).catch(() => null)
  ])
  const c = air.current ?? {}
  const now: string = c.time ?? ''
  const h = air.hourly ?? { time: [] }
  const history: HistoryPoint[] = (h.time as string[])
    .map((t, i) => ({ time: t, aqi: num(h.us_aqi?.[i]), pm25: num(h.pm2_5?.[i]), pm10: num(h.pm10?.[i]) }))
    .filter(x => x.time <= now && (x.aqi != null || x.pm25 != null || x.pm10 != null))
    .slice(-24)
  return {
    isDemo: false, source: 'Open-Meteo Air Quality API (CAMS model estimate, not a ground sensor)', timezone: air.timezone ?? '', time: now,
    aqi: num(c.us_aqi), pm25: num(c.pm2_5), pm10: num(c.pm10), co: num(c.carbon_monoxide), no2: num(c.nitrogen_dioxide), o3: num(c.ozone),
    temp: num(wx?.current?.temperature_2m), humidity: num(wx?.current?.relative_humidity_2m), wind: num(wx?.current?.wind_speed_10m), history
  }
}

export type LevelKey = 'good' | 'moderate' | 'sensitive' | 'unhealthy' | 'very' | 'hazardous'
export interface Level { key: LevelKey; label: string; min: number; max: number; color: string; text: string }

  { key: 'moderate', label: 'Moderate', min: 51, max: 100, color: '#EAB308', text: 'Some pollutants may affect unusually sensitive individuals.' },
  { key: 'sensitive', label: 'Unhealthy for Sensitive Groups', min: 101, max: 150, color: '#F59E0B', text: 'People with lung or heart conditions, children, and older adults may experience health effects. Most people are less likely to be affected.' },
  { key: 'unhealthy', label: 'Unhealthy', min: 151, max: 200, color: '#EF4444', text: 'Outdoor exposure should be reduced. Sensitive groups may experience more serious effects.' },
  { key: 'very', label: 'Very Unhealthy', min: 201, max: 300, color: '#8B5CF6', text: 'Health precautions are recommended for everyone, and sensitive groups should avoid outdoor exertion.' },

export const getLevel = (aqi: number): Level => LEVELS.find(l => aqi <= l.max) ?? LEVELS[LEVELS.length - 1]
export const levelTier = (k: LevelKey | null): 0 | 1 | 2 => (k === 'very' || k === 'hazardous' ? 2 : k === 'sensitive' || k === 'unhealthy' ? 1 : 0)

// PM2.5 concentration bands (µg/m³), US EPA breakpoints (2024). Used only for station markers,
// labelled as "PM2.5 band" – never presented as an AQI value.

export function pm25Band(v: number): Level {
  const key = PM25_BANDS.find(([max]) => v <= max)?.[1] ?? 'hazardous'
  return LEVELS.find(l => l.key === key)!
}

import type { Place } from './api'
export type CategoryKey = 'urban-road' | 'industrial' | 'construction' | 'residential' | 'agricultural' | 'coastal' | 'green'
export interface EnvResult { category: CategoryKey | null; features: { key: CategoryKey; count: number }[]; radius: number }
const RADIUS = 500
const query = (p: Place) => `[out:json][timeout:20];(
way(around:${RADIUS},${p.lat},${p.lon})["highway"~"^(motorway|trunk|primary)$"];
nwr(around:${RADIUS},${p.lat},${p.lon})["landuse"~"^(industrial|construction|residential|farmland|farmyard|orchard|vineyard|meadow|forest|grass|recreation_ground)$"];
nwr(around:${RADIUS},${p.lat},${p.lon})["leisure"~"^(park|garden|nature_reserve)$"];
nwr(around:${RADIUS},${p.lat},${p.lon})["natural"~"^(coastline|beach|wood)$"];
nwr(around:${RADIUS},${p.lat},${p.lon})["man_made"="works"];
);out tags 200;`
const tie: Record<CategoryKey, number> = { industrial: 3, construction: 3, 'urban-road': 2, coastal: 2, agricultural: 1, residential: 1, green: 1 }

function classify(t: Record<string, string>): CategoryKey | null {
  if (t.highway) return 'urban-road'
  if (t.landuse === 'industrial' || t.man_made === 'works') return 'industrial'
  if (t.landuse === 'construction') return 'construction'
  if (t.landuse === 'residential') return 'residential'
  if (['farmland', 'farmyard', 'orchard', 'vineyard', 'meadow'].includes(t.landuse)) return 'agricultural'
  if (t.natural === 'coastline' || t.natural === 'beach') return 'coastal'
  if (t.leisure || t.natural === 'wood' || ['forest', 'grass', 'recreation_ground'].includes(t.landuse)) return 'green'
  return null
}

// Uses OpenStreetMap data via the Overpass API. The result describes what is MAPPED within ~500 m,
// not what is actually polluting the air.
export async function detectEnvironment(p: Place): Promise<EnvResult> {
  const r = await fetch('https://overpass-api.de/api/interpreter', { method: 'POST', body: 'data=' + encodeURIComponent(query(p)) })
  if (!r.ok) throw new Error(`Overpass request failed (${r.status})`)
  const j = await r.json()
  const counts = new Map<CategoryKey, number>()
  for (const el of j.elements ?? []) { const k = classify(el.tags ?? {}); if (k) counts.set(k, (counts.get(k) ?? 0) + 1) }
  const features = [...counts].map(([key, count]) => ({ key, count })).sort((a, b) => b.count - a.count || tie[b.key] - tie[a.key])
  return { category: features[0]?.key ?? null, features, radius: RADIUS }
}

export interface Plan { before: string[]; during: string[]; after: string[]; note: string }

const CATEGORY_EXTRA: Record<CategoryKey, { before?: string; during?: string; after?: string }> = {
  'urban-road': { before: 'Plan a route along side streets or parks instead of main roads.', during: 'Keep distance from idling vehicles and busy junctions.' },
  industrial: { before: 'Check local warnings for smoke, odour, or emissions incidents.', during: 'Move upwind or indoors if you notice smoke or a strong chemical smell.' },
  construction: { during: 'Stay away from dusty work zones and cross to the upwind side.', after: 'Wash your face and rinse your nose or eyes if dusty.' },
  residential: { during: 'Avoid areas where waste or leaves are being burned.' },
  agricultural: { before: 'Check for field burning, spraying, or pollen alerts.', after: 'Change out of dusty clothes before entering the home.' },
  coastal: { before: 'Check whether a port or industrial zone is upwind.' },
  green: { during: 'Pick shaded paths away from busy roads; check the pollen forecast if you have allergies.' }
}

export function buildPlan(level: LevelKey | null, category: CategoryKey | null, sensitive: boolean): Plan {
  const tier = levelTier(level)
  const before = ['Check the latest air quality for your location.']
  const during: string[] = []
  const after: string[] = []

  if (tier === 0) {
    before.push('No special precautions are needed for most people. Keep any prescribed medicine with you if you have a respiratory condition.')
    during.push('Normal outdoor activity is generally fine. Still avoid standing next to heavy traffic or smoke.')
    after.push('Rinse off if you were in a dusty area.', 'Pay attention to coughing, wheezing, or shortness of breath.')
  } else if (tier === 1) {
    before.push('Plan a shorter or lower-intensity outing and pick a route away from busy roads.', 'Prepare a well-fitting N95/FFP2-type respirator if you must be outside for long. Cloth and loose surgical masks filter fine particles poorly.')
    during.push('Reduce strenuous outdoor exercise and take breaks indoors.', 'Avoid heavily polluted areas such as busy junctions and smoke.', 'Use a properly fitted respirator when appropriate; check that it seals around the nose and chin.')
    after.push('Wash exposed skin and face, and change dusty clothing.', 'Watch for cough, wheeze, chest tightness, or throat irritation over the next hours.')
  } else {
    before.push('Consider postponing non-essential outdoor activities and check official health and environmental warnings.', 'Prepare a well-fitting N95/FFP2-type respirator if you must go out.')
    during.push('Keep outdoor time as short and gentle as possible; avoid strenuous activity outdoors.', 'Use a properly fitted respirator and stay away from pollution sources.')
    after.push('Wash exposed skin and face, change clothes, and rinse your nose or eyes if irritated.', 'Monitor symptoms closely and follow official health advice.')
  }
  after.push('Seek medical advice if symptoms are concerning or persistent. Get urgent care for severe breathing difficulty or chest pain.')

  if (sensitive) {
    before.push('If you have a respiratory or heart condition, follow your doctor’s action plan and carry prescribed medicine.')
    if (tier >= 1) during.push('Take extra care: if you are in a sensitive group, limit time outdoors even more than the general advice.')
  }
  const x = category ? CATEGORY_EXTRA[category] : undefined
  if (x?.before) before.push(x.before)
  if (x?.during) during.push(x.during)
  if (x?.after) after.push(x.after)

  return { before, during, after, note: 'General health information, not medical advice. Individual needs differ; ask a health professional about your situation.' }
}

            <p className="mb-4 text-sm text-ink-muted">For <b className="text-ink">{placeLabel(place)}</b>: {level ? <>{AQI_NAME} {Math.round(air!.aqi!)} ({level.label})</> : 'air index unavailable'}{cat && <> · {CATEGORIES[cat].name}</>}{air!.isDemo && <b className="text-warn"> · DEMO MODE</b>}</p>
            <PreventionRecommendation plan={plan} />
            {!level && <p className="mt-3 text-sm text-ink-muted">The air index is unavailable, so the advice above is general.</p>}
          </>
        ) : status === 'loading' ? <Card className="text-sm text-ink-muted">Loading your location’s data…</Card>
          : <Card className="text-sm text-ink-muted">Choose a location to get advice that matches its air quality. In the meantime, here is general guidance for good air:<div className="mt-4"><PreventionRecommendation plan={buildPlan(null, null, sensitive)} /></div></Card>}
      </div>

      <h2 className="mt-14 text-2xl font-bold">Prevention by environment type</h2>
      <div className="mt-4 grid gap-4 md:grid-cols-2">{CATEGORY_LIST.map(c => <EnvironmentalCategoryCard key={c.key} category={c} compact />)}</div>
    </div>
  )
}
